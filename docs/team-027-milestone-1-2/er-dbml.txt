  MENTOR
  ADMIN
}

  DIPLOMA
  DEGREE
}

  PENDING
  APPROVED
  REJECTED
}

  VENTURE
  UNSURE
}

  SUBMITTED
  UNDER_REVIEW
  WAITING

  CLOSED
}

Enum check_status {
  PENDING
  CONFIRMED
  DECLINED
  EXPIRED
  USED
}

  ACTIVE
  CLOSED
}

  COMPLETED
  REASSIGNED
  STUDENT_WITHDREW
  MENTOR_UNAVAILABLE

  SCHEDULED

// 2. TABLES

Table User {
  id integer [pk, increment]
  email varchar(255) [unique, not null]
  password varchar(128) [not null]
  first_name varchar(150)
  last_name varchar(150)
  role user_role [not null, default: 'STUDENT']
  is_active boolean [default: true]
  is_staff boolean [default: false]
  date_joined datetime [not null]
}

Table StudentProfile {
  id integer [pk, increment]
  user_id integer [not null, unique]
  iitm_roll_number varchar(50) [not null]
  level student_level [not null]
  courses json
  cgpa decimal(4,2)
  approval_status profile_approval_status [default: 'PENDING']
  rejection_reason text
  approved_by_id integer
  approved_at datetime
  created_at datetime [not null]
}

Table MentorProfile {
  id integer [pk, increment]
  user_id integer [not null, unique]
  alumni_roll_number varchar(50)
  graduation_year integer
  domain varchar(150) [not null]
  specialty varchar(200) [not null]
  current_role varchar(150) [not null]
  company varchar(150)
  career_path varchar(150)
  experience_years integer [default: 0]
  is_verified boolean [default: false]
  is_available boolean [default: true]
  invited_by_id integer
  bio varchar(500)
  skills json
  linkedin_url varchar(300)
  is_discoverable boolean [default: false]
  verified_by_id integer
  verified_at datetime
  created_at datetime [not null]
}

Table MentoringRequest {
  id integer [pk, increment]
  student_id integer [not null]
  pathway request_pathway [not null]
  domain varchar(150) [not null]
  background text
  problem_statement text [not null]
  status request_status [default: 'SUBMITTED']
  preferred_mentor_id integer
  submitted_at datetime [not null]
  updated_at datetime [not null]
}

Table AvailabilityCheck {
  id integer [pk, increment]
  request_id integer [not null]
  mentor_id integer [not null]
  status check_status [default: 'PENDING']
  sent_at datetime [not null]
  expires_at datetime [not null]
  responded_at datetime
  reason varchar(500)
}

Table Mentorship {
  id integer [pk, increment]
  request_id integer [not null]
  availability_check_id integer [not null, unique]
  student_id integer [not null]
  mentor_id integer [not null]
  approved_by_id integer [not null]
  status mentorship_status [default: 'ACTIVE']
  started_at datetime [not null]
  closed_at datetime
  closing_reason text
  closure_type closure_type
  closed_by_id integer
}

Table Goal {
  id integer [pk, increment]
  mentorship_id integer [not null]
  slot integer [not null, note: '1 to 3']
  title varchar(250) [not null]
  target_date date [not null]
  completed boolean [default: false]
  completed_at datetime
}

Table Meeting {
  id integer [pk, increment]
  mentorship_id integer [not null]
  session_date date [not null]
  duration_minutes integer [not null]
  status meeting_status [default: 'HELD']
  logged_by_id integer [not null]
  mentor_confirmed boolean [default: false]
  start_at datetime
  end_at datetime
  timezone varchar(64)
  title varchar(120)
  meeting_format meeting_format [default: 'VIDEO']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table Feedback {
  id integer [pk, increment]
  mentorship_id integer [not null]
  author_id integer [not null]
  kind feedback_kind [not null]
  text text [not null]
  created_at datetime [not null]
}

Table Outcome {
  id integer [pk, increment]
  mentorship_id integer [not null]
  kind outcome_kind [not null]
  achieved_on date [not null]
  note varchar(500)
  created_at datetime [not null]
}

Table Notification {
  id integer [pk, increment]
  recipient_id integer [not null]
  title varchar(200) [not null]
  message text [not null]
  is_read boolean [default: false]
  created_at datetime [not null]
}

Table AuditLog {
  id integer [pk, increment]
  actor_id integer
  action varchar(80) [not null]
  entity_type varchar(80) [not null]
  entity_id integer [not null]
  details json
  created_at datetime [not null]
}

Table Alert {
  id integer [pk, increment]
  mentorship_id integer [not null]
  kind varchar(40) [not null, note: 'INACTIVITY | FEEDBACK_DUE']
  period_start date [not null]
  resolved_at datetime
  created_at datetime [not null]
}

Table ActivityStatusHistory {
  id integer [pk, increment]
  request_id integer
  mentorship_id integer
  event varchar(40) [not null]
  from_status varchar(40)
  to_status varchar(40) [not null]
  actor_id integer
  actor_role varchar(20)
  note varchar(300)
  visible_to_student boolean [default: true]
  created_at datetime [not null]
}

Table GoogleCalendarConnection {
  id integer [pk, increment]
  user_id integer [not null, unique]
  status calendar_status [default: 'DISCONNECTED']
  google_email varchar(254)
  scopes varchar(500)
  token_expires_at datetime
  connected_at datetime
  last_error varchar(300)
  updated_at datetime [not null]
}

Table MeetingCalendarEvent {
  id integer [pk, increment]
  meeting_id integer [not null, unique]
  provider varchar(20) [default: 'GOOGLE']
  calendar_owner_id integer
  event_id varchar(255)
  html_link varchar(500)
  meet_link varchar(500)
  sync_status event_sync_status [default: 'SYNCED']
  last_synced_at datetime
  last_error varchar(300)
}

Table KnowledgeArticle {
  id integer [pk, increment]
  title varchar(180) [not null]
  content text [not null]
  audience article_audience [default: 'ALL']
  is_published boolean [default: false]
  updated_at datetime [not null]
}

Table MatchSuggestionAudit {
  id integer [pk, increment]
  request_id integer [not null]
  requested_by_id integer [not null]
  algorithm varchar(30) [default: 'rules-v1']
  candidate_count integer [default: 0]
  used_local_llm boolean [default: false]
  created_at datetime [not null]
}

Table AIRequestAssessment {
  id integer [pk, increment]
  request_id integer [not null]
  suggested_priority priority_level [not null]
  explanation varchar(500) [not null]
  confidence decimal(3,2)
  factors json
  algorithm varchar(30) [default: 'priority-rules-v1']
  override_priority priority_level
  override_reason varchar(300)
  overridden_by_id integer
  overridden_at datetime
  created_at datetime [not null]
}
